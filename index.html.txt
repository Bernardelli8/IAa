<script>
    const cores = ["#7dd3fc", "#86efac", "#fde047", "#fda4af", "#c4b5fd", "#fdba74"];

    // Quebra o texto em pedaços parecidos com tokens reais (sílabas/blocos de ~4 caracteres + pontuação)
    function tokenizar(texto) {
        if (!texto.trim()) return [];
        const partes = texto.match(/\S+\s*/g) || [];
        const tokens = [];
        partes.forEach(parte => {
            for (let i = 0; i < parte.length; i += 4) {
                tokens.push(parte.slice(i, i + 4));
            }
        });
        return tokens;
    }

    function usarExemplo(tipo) {
        const campo = document.getElementById("prompt");
        if (tipo === "curto") {
            campo.value = "Oi, tudo bem?";
        } else if (tipo === "medio") {
            campo.value = "Explique em poucas palavras como funciona a criptografia na internet.";
        } else {
            campo.value = "Atue como um professor de tecnologia e crie um resumo detalhado explicando o que é inteligência artificial, como funcionam os modelos de linguagem, por que prompts maiores gastam mais tokens e dê três exemplos práticos do dia a dia.";
        }
        simularIA();
    }

    function gerarRespostaSimulada(prompt, qtdTokens) {
        const p = prompt.toLowerCase();
        if (p.includes("oi") || p.includes("olá")) {
            return "Olá! Como seu prompt foi bem curto (" + qtdTokens + " tokens), minha resposta também é rápida e econômica.";
        }
        if (qtdTokens <= 10) {
            return "Resposta objetiva: Você enviou um pedido curto, consumindo poucos tokens de processamento.";
        }
        if (qtdTokens <= 30) {
            return "Resposta intermediária: Analisei seu prompt de " + qtdTokens + " tokens. Como você deu mais contexto, gerei uma explicação mais completa sobre o tema solicitado.";
        }
        return "Resposta detalhada: Como o seu prompt foi extenso (" + qtdTokens + " tokens), o modelo precisou processar várias instruções e gerar uma resposta bem mais longa — o que aumenta significativamente o consumo total de tokens.";
    }

    function simularIA() {
        const prompt = document.getElementById("prompt").value;
        if (!prompt.trim()) {
            alert("Digite um prompt primeiro!");
            return;
        }

        const tokensEntrada = tokenizar(prompt);
        const resposta = gerarRespostaSimulada(prompt, tokensEntrada.length);
        const tokensSaida = tokenizar(resposta);
        const total = tokensEntrada.length + tokensSaida.length;

        // Atualiza números
        document.getElementById("tokensPrompt").textContent = tokensEntrada.length;
        document.getElementById("tokensResposta").textContent = tokensSaida.length;
        document.getElementById("tokensTotal").textContent = total;
        document.getElementById("qtdTokensTexto").textContent = tokensEntrada.length;

        // Exibe o prompt digitado e a resposta
        document.getElementById("promptExibido").textContent = prompt;
        document.getElementById("respostaIA").textContent = resposta;

        // Mostra os blocos coloridos de cada token
        const containerTokens = document.getElementById("tokensVisuais");
        containerTokens.innerHTML = "";
        tokensEntrada.forEach((tk, index) => {
            const span = document.createElement("span");
            span.className = "token-chip";
            span.style.background = cores[index % cores.length];
            span.textContent = tk.replace(/ /g, "·");
            containerTokens.appendChild(span);
        });

        // Atualiza barra de consumo
        const barra = document.getElementById("barraConsumo");
        const nivel = document.getElementById("nivelConsumo");
        const porcentagem = Math.min((total / 150) * 100, 100);
        barra.style.width = porcentagem + "%";

        if (tokensEntrada.length <= 10) {
            barra.style.background = "#22c55e";
            nivel.textContent = "Consumo BAIXO — prompt curto economiza bastante tokens.";
        } else if (tokensEntrada.length <= 30) {
            barra.style.background = "#eab308";
            nivel.textContent = "Consumo MÉDIO — prompt moderado usa quantidade intermediária de tokens.";
        } else {
            barra.style.background = "#ef4444";
            nivel.textContent = "Consumo ALTO — prompt grande usa muito mais tokens na entrada e na resposta.";
        }

        document.getElementById("resultado").style.display = "block";
    }
</script>
